function videoEl() {
  if (VID.el) return VID.el;
  const v = document.createElement('video');
  v.id = 'pvVideo'; v.muted = true; v.playsInline = true; v.preload = 'metadata'; v.setAttribute('playsinline', ''); v.setAttribute('aria-hidden', 'true'); v.tabIndex = -1;
  v.addEventListener('loadeddata', () => { VID.ready = true; VID.err = false; });
  v.addEventListener('error', () => { VID.err = true; VID.ready = false; FILM.playing = false; });
  v.addEventListener('ended', () => { FILM.playing = false; });
  document.body.appendChild(v); VID.el = v;
  return v;
}
function setVideoSrc(url) {
  if (url === VID.url) return;
  const v = videoEl(); VID.url = url; VID.ready = false; VID.err = false;
  try { v.pause(); } catch (e) { /* ignore */ }
  if (url) { v.src = url; try { v.load(); } catch (e) { /* ignore */ } } else { v.removeAttribute('src'); try { v.load(); } catch (e) { /* ignore */ } FILM.playing = false; }
}

function monMode() {
  if (!STATE || !STATE.project) return 'empty';
  const st = Pipe.status, mp4 = vidOK();
  if (FILM.user) return mp4 ? 'video' : 'film';
  if (st === 'queued') return 'card';
  if (st === 'running' || st === 'paused') {
    const id = Pipe.active;
    if (id === 'director') return TLS.beatRev < 1 && BEATS.length ? 'listen' : 'board';
    if (id === 'reference') return BEATS.length ? 'listen' : 'card';
    if (id === 'editor') return 'weave';
    if (id === 'audio') return 'mix';
    if (id === 'motion') return 'code';
    if (id === 'qa') return 'render';
    return 'card';
  }
  if (st === 'failed' || st === 'stopped') return mp4 && FILM.playing ? 'video' : 'halt';
  if (st === 'completed') return mp4 ? 'video' : SCENES.length ? 'film' : 'poster';
  if (FILM.playing) return mp4 ? 'video' : 'film';
  return mp4 ? 'video' : 'poster';
}
