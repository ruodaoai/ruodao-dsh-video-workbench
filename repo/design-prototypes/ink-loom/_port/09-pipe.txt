        s.tok = { key: fl.key, x: S.x, y: S.y, vx: 0, vy: -40, mode: 'hang' }; if (s.hook) s.hook.u = 1;
      }
      if (fl.kind === 'fetch' && te >= T.carry1 && !ev.absorb) { ev.absorb = 1; const P0 = shDeck(s); if (s.tok) { s.tok.mode = 'merge'; s.tok.m0 = SIM.t; s.tok.mx = s.tok.x; s.tok.my = s.tok.y; s.tok.tx = P0.x; s.tok.ty = P0.y; } burst(P0.x, P0.y, 16, { col: AG[fl.agent].glaze, sp: 80, life: 0.8 }); s.hook = null; AG[fl.agent].flash = 1; }
      if (fl.kind === 'weave') {
        const tg = targetPts(fl);
        if (te >= T.dive1 + fl.sw * 0.2 && !ev.merge) { ev.merge = 1; if (s.tok) { s.tok.mode = 'merge'; s.tok.m0 = SIM.t; s.tok.mx = s.tok.x; s.tok.my = s.tok.y; s.tok.tx = tg.cx; s.tok.ty = tg.y; } }
        if (te >= T.dive1 && !ev.stitch) { ev.stitch = 1; TLSTITCH.push({ ta: fl.target.t0, tb: fl.target.t1, row: fl.target.track, t0: SIM.t }); }
        if (te >= T.land && !ev.land) {
          ev.land = 1; s.hook = null;
          if (fl.target.track === 'shot') { const q = CLIP[fl.target.sc]; if (q) { q.st = Math.max(q.st, 2); q.fx = SIM.t; q.landed = true; } }
          else TLS.bgmLanded = Pipe.runId;
          A.woven = Pipe.runId; syncUsedBadges();
          bloom(tg.cx, tg.y, fl.target.track === 'bgm' ? 70 : 46);
          burst(tg.cx, tg.y, 22, { sp: 140, life: 1.1, sz: 3 });
          CAM.shake = Math.max(CAM.shake, 0.16);
        }
      }
    }
    if (te >= T.end) {
      FLIGHTS.splice(i, 1);
      if (s.fl === fl) {
        s.fl = null; s.hook = null;
        if (fl.kind === 'weave') startReturn(fl.agent, 0.95);
        else { const p = toStage(s.dx, s.dy); s.x = p.x; s.y = p.y; s.vx = s.vy = 0; s.trail.length = 0; }
      }
    }
  }
}
function curFlight() { for (const f of FLIGHTS) if (f.kind === 'weave') return f; return null; }
