  const span = clamp((TLV.a1 - TLV.a0) / f, Math.min(6, DUR + 1.2), DUR + 1.2), c = at == null ? (TLV.a0 + TLV.a1) / 2 : at;
  const u = at == null ? 0.5 : (at - TLV.a0) / (TLV.a1 - TLV.a0);
  let a0 = c - span * u; a0 = clamp(a0, -0.6, DUR + 0.6 - span);
  TLV.a0 = a0; TLV.a1 = a0 + span; TLV.manual = true; $('#tlFit').classList.toggle('on', span >= DUR);
}
