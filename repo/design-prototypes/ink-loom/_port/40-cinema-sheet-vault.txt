/* 全屏观看：成片铺满整个窗口（面板内核 PK.cinema，再按 F 进入真正的全屏）；关闭后监视器从同一时刻继续 */
function openCinema() {
  if (!hasProject()) { toast('请先新建或打开一个视频工程', '卷', '新建工程', () => openKit('newProject'), true); return; }
  if (!PREVIEW.hasMp4 || !PREVIEW.url) { toast(Pipe.live() ? '织造进行中 · 出片完成后即可全屏观看' : '还没有成片 · 一键出片完成后即可全屏观看', '片'); return; }
  if (!window.PK || typeof PK.cinema !== 'function') { toast('面板内核未加载，请刷新页面后重试', '断', null, null, true); return; }
  const at = vidOK() ? VID.el.currentTime || 0 : 0;
  if (FILM.playing) toggleFilm();
  PK.cinema({
    src: PREVIEW.url, poster: PREVIEW.poster || '', title: PROJECT.title || '成片', sub: `${PREVIEW.dur ? fmtT(PREVIEW.dur) + ' · ' : ''}${PROJECT.w}×${PROJECT.h}`, startAt: at,
    onClose: ({ time }) => { seekPreview(time * (DUR / Math.max(0.01, vidDur()))); },
  });
}
