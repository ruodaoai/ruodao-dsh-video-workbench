function focusPoint() {
  if (CAM.focus) { const h = L.ag[CAM.focus]; return { x: h.x, y: h.y + (L.tiny ? 34 : 46), s: 1.75 }; }
  const st = Pipe.status, id = Pipe.active, a = AG[id];
  if (Pipe.live()) {
    if (!a || st === 'queued') return { x: SW.cx, y: L.beamY + 110, s: 1.22 };
    if (a.status === 'err') return { x: L.ag[id].x, y: L.ag[id].y + 50, s: 1.85 };
    if (id === 'director' && TLS.beatRev >= 1) return { x: L.mon.cx, y: L.mon.cy - 12, s: 1.34 };
    if (id === 'audio' || id === 'motion' || id === 'qa') return { x: L.mon.cx, y: L.mon.cy - 12, s: 1.36 };
    if (id === 'editor' && curFlight()) return { x: L.ag.editor.x, y: L.ag.editor.y + 70, s: 1.36 };
    const h = L.ag[id]; return h ? { x: h.x, y: h.y + (L.tiny ? 40 : 56), s: 1.6 } : { x: SW.cx, y: SW.cy, s: 1.2 };
  }
  if (st === 'failed' && L.ag[id]) return { x: L.ag[id].x, y: L.ag[id].y + 50, s: 1.7 };
  if (st === 'completed' || PREVIEW.hasMp4) return { x: L.mon.cx, y: L.mon.cy + 18, s: 1.26 };
  return { x: SW.cx + Math.sin(SIM.t * 0.07) * SW.w * 0.2, y: L.beamY + 150, s: 1.32 };
}

const TIP = $('#tip');
function showTip(html, x, y) {
  TIP.innerHTML = html; TIP.classList.add('on');
  const r = TIP.getBoundingClientRect(); let tx = x + 16, ty = y + 16;
  if (tx + r.width > innerWidth - 8) tx = x - r.width - 16; if (ty + r.height > innerHeight - 8) ty = y - r.height - 16;

function toast(msg, ch = '织', actLabel, actFn, warn) {
  const box = $('#toasts'), t = mk('div', 'toast' + (warn ? ' warn' : ''));
  t.setAttribute('role', warn ? 'alert' : 'status');
  const ic = mk('i'); ic.textContent = ch; t.appendChild(ic);
  const sp = mk('span'); sp.textContent = String(msg == null ? '' : msg); t.appendChild(sp);
  if (actLabel) { const b = mk('button', 'btn xs gold'); b.textContent = actLabel; b.setAttribute('aria-label', actLabel); b.onclick = () => { try { actFn(); } finally { t.remove(); } }; t.appendChild(b); }
  const x = mk('button', 't-x', '<svg><use href="#i-x"/></svg>'); x.setAttribute('aria-label', '关闭提示'); x.onclick = () => t.remove(); t.appendChild(x);
  box.appendChild(t);
  while (box.children.length > 3) box.firstChild.remove();
  setTimeout(() => { t.classList.add('out'); setTimeout(() => t.remove(), 420); }, actLabel ? 7000 : warn ? 6000 : 3600);
}
