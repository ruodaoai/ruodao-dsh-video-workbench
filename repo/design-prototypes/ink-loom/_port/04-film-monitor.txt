  for (let x = x0; x < rv; x += 2 * k) { const t = ((x - x0) / (x1 - x0)) * DUR, v = envMax(BGM_ENV, t - 0.2, t + 0.2) * amp; c.fillRect(x, mid - v, 1.3 * k, v * 2); }
  c.fillStyle = 'rgba(237,227,207,.14)';
  for (let x = rv; x < x1; x += 2 * k) { const t = ((x - x0) / (x1 - x0)) * DUR, v = envMax(BGM_ENV, t - 0.2, t + 0.2) * amp * 0.6; c.fillRect(x, mid - v, 1.3 * k, v * 2); }
  BEATS.forEach(([tb, st]) => { const x = lerp(x0, x1, tb / DUR); if (x > rv) return; c.fillStyle = st ? 'rgba(242,213,146,.9)' : 'rgba(242,213,146,.32)'; c.fillRect(x, mid - amp - (st ? 12 : 7) * k, st ? 1.6 * k : 1 * k, (st ? 9 : 5) * k); });
  DROPS.forEach((td, i) => { const x = lerp(x0, x1, td / DUR); if (x > rv) return; c.fillStyle = '#C8361D'; c.beginPath(); c.moveTo(x, mid + amp + 4 * k); c.lineTo(x - 5 * k, mid + amp + 12 * k); c.lineTo(x + 5 * k, mid + amp + 12 * k); c.closePath(); c.fill(); c.font = `${Math.max(8, Math.round(9 * k))}px KaiTi, serif`; c.textAlign = 'center'; c.fillText('落' + (i + 1), x, mid + amp + 22 * k); });
  c.strokeStyle = '#F2D592'; c.lineWidth = 1.4 * k; c.beginPath(); c.moveTo(rv, mid - amp - 14 * k); c.lineTo(rv, mid + amp + 6 * k); c.stroke();
  c.textAlign = 'left'; c.textBaseline = 'alphabetic'; c.fillStyle = '#F2D592'; c.font = `${Math.round(34 * k)}px KaiTi, serif`;
  const bpmTxt = PROJECT.bpm ? String(PROJECT.bpm) : '—';
  c.globalAlpha = clamp(pr * 3, 0.25, 1); c.fillText(bpmTxt, 18 * k, 46 * k); c.globalAlpha = 1;
  c.fillStyle = 'rgba(237,227,207,.6)'; c.font = `${Math.max(8, Math.round(10 * k))}px Consolas, "Microsoft YaHei UI", monospace`; c.fillText('BPM · 节拍分析', 82 * k, 44 * k);
  const rvT = pr * DUR, strong = BEATS.filter((b) => b[1] && b[0] <= rvT).length, drops = DROPS.filter((d) => d <= rvT).length;
  c.fillText(fitText(c, `${BEATS.length ? '首拍 ' + BEATS[0][0].toFixed(3) + 's · ' : ''}强拍 ${strong} · 突降 ${drops}/${DROPS.length}`, w * 0.6), 18 * k, 62 * k);
  c.textAlign = 'right'; c.fillText(fitText(c, MUSIC[0] ? MUSIC[0].name : '配乐分析', w * 0.36), w - 18 * k, 26 * k);
}
function boardGrid(N, w, h) {
  let best = null;
  for (let cols = 1; cols <= 12; cols++) { const rows = Math.ceil(N / cols), cw = w / cols, ch = h / rows, sc = Math.min(cw, (ch * 16) / 9); if (!best || sc > best.s + 0.01) best = { cols, rows, s: sc }; }
  return best;
}
function renderBoard(c, w, h, n, t) {
  c.fillStyle = '#0D0C0A'; c.fillRect(0, 0, w, h);
  const N = SCENES.length, k = w / 520, hb = 22 * k, gap = 3 * k;
  c.font = `${Math.max(8, Math.round(10 * k))}px Consolas, "Microsoft YaHei UI", monospace`; c.textBaseline = 'middle'; c.textAlign = 'left'; c.fillStyle = 'rgba(237,227,207,.7)';
  c.fillText(fitText(c, `分镜 STORYBOARD · ${n}/${N} · ${PROJECT.w}×${PROJECT.h}`, w * 0.7), 8 * k, hb / 2 + 1);
  c.textAlign = 'right'; c.fillStyle = '#D4A64A'; c.fillText('策划导演', w - 8 * k, hb / 2 + 1); c.textAlign = 'left';
  if (!N) { c.textAlign = 'center'; c.fillStyle = 'rgba(237,227,207,.45)'; c.font = `${Math.max(10, Math.round(14 * k))}px KaiTi, serif`; c.fillText('策划导演正在拆解叙事 · 分镜即将落笔', w / 2, h / 2); return; }
  const g = boardGrid(N, w - gap, h - hb - gap), cols = g.cols, rows = g.rows;
  const cw = (w - gap * (cols + 1)) / cols, ch = (h - hb - gap * (rows + 1)) / rows;
  for (let i = 0; i < N; i++) {
    const x = gap + (i % cols) * (cw + gap), y = hb + gap + Math.floor(i / cols) * (ch + gap);
    if (i < n) {
      const age = clamp((n - i) / 3, 0, 1), sc = lerp(1.12, 1, E.out(age));
      c.save(); c.translate(x + cw / 2, y + ch / 2); c.scale(sc, sc); c.globalAlpha = 0.35 + 0.65 * age;
      drawSceneCard(c, i, -cw / 2, -ch / 2, cw, ch);
      c.globalAlpha = 1; c.strokeStyle = i === n - 1 ? '#F2D592' : 'rgba(212,166,74,.35)'; c.lineWidth = i === n - 1 ? 1.5 : 0.8; c.strokeRect(-cw / 2, -ch / 2, cw, ch);
      if (cw > 22 * k) { c.fillStyle = 'rgba(0,0,0,.55)'; c.fillRect(-cw / 2, -ch / 2, 15 * k, 9 * k); c.fillStyle = '#F2E9D8'; c.font = `${Math.max(6, Math.round(7 * k))}px Consolas, monospace`; c.fillText(String(i + 1).padStart(2, '0'), -cw / 2 + 2 * k, -ch / 2 + 4.8 * k); }
      c.restore();
    } else { c.strokeStyle = 'rgba(237,227,207,.12)'; c.setLineDash([3, 3]); c.lineWidth = 0.8; c.strokeRect(x + 0.5, y + 0.5, cw - 1, ch - 1); c.setLineDash([]); }
  }
}
