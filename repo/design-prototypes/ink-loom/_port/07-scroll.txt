function waveShape(c, arr, ta, tb, R, amp, t, wob) {
  const r = L.tlc, xa = Math.max(r.x, tX(ta)), xb = Math.min(r.x + r.w, tX(tb));
  if (xb - xa < 1) return false;
  const mid = R.c, step = 2, top = [];
  c.beginPath();
  for (let x = xa; x <= xb; x += step) { const tt = xT(x), half = (TLV.v1 - TLV.v0) / r.w * step; let v = envMax(arr, tt - half, tt + half); v *= amp * (1 + (wob ? 0.07 * Math.sin(x * 0.08 + t * 5.2) : 0)); top.push(x, v); x === xa ? c.moveTo(x, mid - v) : c.lineTo(x, mid - v); }
  for (let i = top.length - 2; i >= 0; i -= 2) c.lineTo(top[i], mid + top[i + 1] * 0.92);
  c.closePath();
  return true;
}
