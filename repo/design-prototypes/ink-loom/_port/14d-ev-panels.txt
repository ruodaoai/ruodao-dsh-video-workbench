function bindPanels() {

    const sl = e.target.closest('.seal-tg[data-k]'); if (sl) { setParam(sl.dataset.k); return; }
    if (e.target.closest('#lowTg')) { act('setLowSpec', { on: !LOWSPEC }); return; }
    const sb = e.target.closest('.sel-btn'); if (sb) { const s = sb.closest('.sel'), was = s.classList.contains('open'); closeMenus(); s.classList.toggle('open', !was); sb.setAttribute('aria-expanded', !was ? 'true' : 'false'); if (!was && s.classList.contains('vsel')) openVoiceMenu(s); e.stopPropagation(); return; }

  });
  $('#foPages').addEventListener('keydown', (e) => {
    const sl = e.target.closest('.seal-tg'); if (sl && (e.key === 'Enter' || e.key === ' ')) { e.preventDefault(); e.stopPropagation(); if (sl.id === 'lowTg') act('setLowSpec', { on: !LOWSPEC }); else setParam(sl.dataset.k); return; }
    const sk = e.target.closest('.silk'); if (sk && (e.key === 'ArrowLeft' || e.key === 'ArrowRight')) { e.preventDefault(); e.stopPropagation(); const k = sk.dataset.k, stp = +sk.dataset.step, mn = +sk.dataset.min, mx = +sk.dataset.max; setParam(k, clamp((+DIR[k] || mn) + (e.key === 'ArrowRight' ? stp : -stp), mn, mx)); }

    const set = (ev) => { const r = s.getBoundingClientRect(), u = clamp((ev.clientX - r.left) / Math.max(1, r.width), 0, 1); v = Math.round((mn + u * (mx - mn)) / stp) * stp; DIR[k] = v; PENDING[k] = { v, until: performance.now() + 2600 }; syncFolio(); };
    set(e); try { s.setPointerCapture(e.pointerId); } catch (x) { /* ignore */ }

  });

function syncCapsuleDraft() { const ta = $('#foTopic'), t = ta ? ta.value.trim() : ''; const cap = $('#capTxt'); cap.textContent = t || '写下创意主题 · 点此展开经折'; cap.classList.toggle('ph', !t); }
function bindKeys() {
  document.addEventListener('keydown', (e) => {
    const k = e.key;
    if (k === 'Escape') { escape(); return; }
    if (e.ctrlKey || e.metaKey || e.altKey) return;
    if (e.target.closest && e.target.closest('input, textarea, select, [contenteditable="true"]')) return;
    if (k === ' ' && e.target.closest && e.target.closest('button, [role="button"], [role="switch"], [role="slider"]')) return;
    if (k === '1' || k === '2' || k === '3') setCam(['overview', 'follow', 'close'][+k - 1]);
    else if (k === ' ') { e.preventDefault(); toggleFilm(); }
    else if (k === 'p' || k === 'P') (FOLIO_OPEN ? closeFolio() : openFolio());
    else if (k === 't' || k === 'T') $('#btnTheme').click();
    else if (k === 'r' || k === 'R') runAction();
    else if (k === '+' || k === '=') tlZoom(1.5, PH.t);
    else if (k === '-' || k === '_') tlZoom(1 / 1.5, PH.t);
    else if (k === 'f' || k === 'F') tlFit();
    else if (k === '?') openModal('help');
  });
}
