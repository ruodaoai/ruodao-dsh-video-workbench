function groupHTML([k, cn, en, kind, a, b, st]) {
  const src = kind === 'slider' || kind === 'seal' ? '' : a;
  const lab = `<label>${cn}<em>${en}</em>${kind === 'slider' ? `<output data-o="${k}"></output>` : ''}</label>`;
  if (kind === 'slider') return `<div class="pg">${lab}<div class="silk" data-k="${k}" data-min="${a}" data-max="${b}" data-step="${st}" role="slider" tabindex="0" aria-label="${cn}" aria-valuemin="${a}" aria-valuemax="${b}"><i></i><b></b></div></div>`;
  if (kind === 'seal') return `<div class="pg">${lab}<div class="seal-tg" data-k="${k}" role="switch" tabindex="0" aria-label="${cn}"><span class="box"><b>启</b></span><span class="tx"><span class="tx-a">成片叠加数字人主播</span><small>口型同步 · 透明画中画</small></span></div></div>`;
  if (kind === 'voice') return `<div class="pg">${lab}<div class="sel vsel" data-k="${k}"><button class="sel-btn" aria-haspopup="listbox" aria-label="${cn}"><span></span><svg><use href="#i-caret"/></svg></button><div class="menu" role="listbox"><input class="vq" type="search" placeholder="搜索音色" aria-label="搜索音色"><div class="vlist"></div></div></div></div>`;
  const items = optsFor(src).map((o) => `<button class="tag-s${kind === 'w' ? ' w' : ''}" data-v="${esc(o.value)}" title="${esc(o.label)}">${kind === 'sw' ? `<i class="sw" style="background:${STYLE_SW[o.value] || 'var(--gold)'}"></i>` : ''}<span>${esc(o.label)}</span></button>`).join('');
  return `<div class="pg">${lab}<div class="tags" data-k="${k}" role="group" aria-label="${cn}">${items || '<span class="lf-note">（宿主未提供选项）</span>'}</div></div>`;
}

function buildFolio() {
  FOLIO_SIG = JSON.stringify(Object.keys(OPTV).map((k) => (OPTV[k] || []).length + ':' + (OPTV[k] || []).slice(0, 40).map((o) => o.value + '=' + o.label).join('|')));
  $('#foPages').innerHTML = FOLIO.map((p, i) => `<section class="leaf" style="--i:${i}"><div class="lf-h"><span>${p.ch}</span><b>${p.cn}</b><i>${p.en}</i></div>${p.g.map(groupHTML).join('')}${p.x ? extraHTML(p.x) : ''}</section>`).join('');
  syncFolio();
}
function folioMaybeRebuild() {
  const sig = JSON.stringify(Object.keys(OPTV).map((k) => (OPTV[k] || []).length + ':' + (OPTV[k] || []).slice(0, 40).map((o) => o.value + '=' + o.label).join('|')));
  if (sig !== FOLIO_SIG && !(FOLIO_OPEN && document.activeElement && $('#folio').contains(document.activeElement))) buildFolio(); else syncFolio();

  $$('#foPages .vsel').forEach((s) => { s.querySelector('.sel-btn span').textContent = optLbl(OPTV.voices, DIR.voiceId); s.querySelector('.sel-btn span').title = optLbl(OPTV.voices, DIR.voiceId); });
  $$('#foPages .silk').forEach((s) => { const k = s.dataset.k, mn = +s.dataset.min, mx = +s.dataset.max, v = clamp(+DIR[k] || mn, mn, mx), u = (v - mn) / (mx - mn); s.querySelector('i').style.width = u * 100 + '%'; s.querySelector('b').style.left = u * 100 + '%'; s.setAttribute('aria-valuenow', String(v)); const o = $(`output[data-o="${k}"]`); if (o) o.textContent = fmtMS(v); s.classList.toggle('dis', k === 'targetDurationSec' && DIR.durationMode !== 'custom'); });
  $$('#foPages .seal-tg[data-k]').forEach((s) => { const on = !!DIR[s.dataset.k]; s.classList.toggle('on', on); s.setAttribute('aria-checked', on ? 'true' : 'false'); });

function setParam(k, v) {
  if (k === 'fps' || k === 'targetDurationSec') v = Number(v);
  if (k === 'avatarEnabled') v = !DIR.avatarEnabled;
  if (String(DIR[k]) === String(v)) return;
  DIR[k] = v; PENDING[k] = { v, until: performance.now() + 2600 };
  syncFolio();
  const cap = $('#capsule'); cap.classList.remove('flash'); void cap.offsetWidth; cap.classList.add('flash');
  act('setDirector', { patch: { [k]: v } });
  if (Pipe.live() && !FOLIO_WARNED) { FOLIO_WARNED = true; toast('参数已同步 · 运行中的这一卷不受影响，下次出片生效', '折'); }
}
