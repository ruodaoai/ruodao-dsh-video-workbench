function sceneAt(t) {
  if (!SCENES.length) return NO_SCENE;
  if (t <= 0) return SCENES[0];
  let lo = 0, hi = SCENES.length - 1;
  while (lo < hi) { const m = (lo + hi + 1) >> 1; if (SCENES[m].t0 <= t) lo = m; else hi = m - 1; }
  return SCENES[lo];
}
