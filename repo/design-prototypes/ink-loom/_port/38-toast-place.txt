const TOAST_KIT = { head: '.pk-frame-h', full: '.pk-frame-idx,.pk-frame-ico,.pk-frame-t,.pk-frame-en', short: '.pk-frame-idx,.pk-frame-ico,.pk-frame-t', end: '.pk-frame-tools>*,.pk-frame-x' };
const TOAST_FOLIO = { head: '.fo-head', full: '.fo-head .pn-t b,.fo-head .pn-t i,.fo-sum', short: '.fo-head .pn-t b,.fo-head .pn-t i', end: '[data-fclose]' };
function placeToasts() {
  const ts = $('#toasts'); if (!ts) return;
  /* 需要让开的「面板」：所有打开的原生面板 + 展开中的经折 */
  const boxes = (window.PK && PK.frames ? PK.frames.stack : []).filter((f) => f && f.el && f.el.isConnected && !f.closed).map((f) => ({ el: f.el, s: TOAST_KIT }));
  const fo = FOLIO_OPEN ? $('#folio .fo-card') : null; if (fo) boxes.push({ el: fo, s: TOAST_FOLIO });
  const put = (o) => {
    ['left', 'right', 'top', 'bottom'].forEach((k) => { ts.style[k] = o[k] == null ? 'auto' : Math.round(o[k]) + 'px'; });
    const w = Math.round(Math.max(200, o.w)) + 'px'; ts.style.maxWidth = w; ts.style.width = o.fixed ? w : '';
  };
  if (!boxes.length) {
    ts.classList.remove('in-kit', 'k-top', 'k-left');
    if (L.vw && SW.w > 1) put({ right: L.vw - SW.x - SW.w + 10, bottom: L.vh - (L.topH + L.tl.y) + 10, w: clamp(SW.x + SW.w - (L.mon.x + L.mon.w) - 20, 240, 440) });
    return;
  }
  let x0 = Infinity, x1 = -Infinity, y0 = Infinity, top = null;
  for (const f of boxes) {
    const r = f.el.getBoundingClientRect(); if (r.width < 4 || r.height < 4) continue;
    x0 = Math.min(x0, r.left); x1 = Math.max(x1, r.right);
    if (r.top < y0) { y0 = r.top; top = f; }
  }
  if (!top) return;
  ts.classList.add('in-kit');
  const roomR = innerWidth - x1 - 28, roomL = x0 - 28;
  if (Math.max(roomR, roomL) >= 280) {
    const right = roomR >= 280;
    ts.classList.remove('k-top'); ts.classList.toggle('k-left', !right);
    put(right ? { right: 14, bottom: 16, w: Math.min(440, roomR + 14) } : { left: 14, bottom: 16, w: Math.min(440, roomL + 14) });
    return;
  }
  ts.classList.remove('k-left'); ts.classList.add('k-top');
  const fr = top.el.getBoundingClientRect(), head = top.el.querySelector(top.s.head);
  let a = fr.left + 16, b = fr.right - 16, y = fr.top + 6;
  if (head) {
    const hr = head.getBoundingClientRect(); y = hr.top + 6;
    const right = (sel) => { let x = hr.left + 14; head.querySelectorAll(sel).forEach((n) => { const r = n.getBoundingClientRect(); if (r.width) x = Math.max(x, r.right); }); return x + 14; };
    let bx = hr.right - 14;
    head.querySelectorAll(top.s.end).forEach((n) => { const r = n.getBoundingClientRect(); if (r.width) bx = Math.min(bx, r.left); });
    bx -= 14;
    /* 优先放在「题名 + 英文细注」与关闭钮之间；太窄就只让开题名（英文细注 / 参数摘要只是装饰） */
    const full = right(top.s.full), short = right(top.s.short);
    if (bx - full >= 420) { a = full; b = bx; } else if (bx - short >= 300) { a = short; b = bx; }
  }
  const w = Math.min(900, b - a);
  put({ left: a + (b - a - w) / 2, top: Math.max(6, y), w, fixed: true });
  while (ts.children.length > 2) ts.firstChild.remove();
  trimTopToasts(ts);
}
/* 页眉里的签条只排一行：换到第二行的旧签条先撤下（最新的一张总会留下） */
function trimTopToasts(box) {
  while (box.children.length > 1 && box.firstElementChild.offsetTop !== box.lastElementChild.offsetTop) box.firstElementChild.remove();
}
let PLACE_T = 0;
function placeToastsSoon() { placeToasts(); clearTimeout(PLACE_T); PLACE_T = setTimeout(placeToasts, 320); }
