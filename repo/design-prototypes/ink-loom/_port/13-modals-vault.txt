function confirmStaged(key) {

  if (card) { card.classList.add('stamped', 'ok'); const tg = card.querySelector('.tag'); if (tg) tg.textContent = '入库中'; }

  });

async function importFiles(fileList) {
  const files = Array.from(fileList || []).filter((f) => f && f.size >= 0).slice(0, 40);
  if (!files.length) return;
  if (!hasProject()) { toast('请先新建或打开一个视频工程', '卷', '新建工程', () => act('newProject', {}), true); return; }
  const total = files.reduce((s, f) => s + (f.size || 0), 0);
  if (total > 2.5 * 1073741824) { toast('单次导入超过 2.5 GB，请分批导入', '卷', null, null, true); return; }
  $('#dropZone').classList.add('busy');
  try {
    const out = [], transfer = [];
    for (const f of files) { const data = await f.arrayBuffer(); out.push({ name: String(f.name || 'upload.bin').replace(/[\\/]/g, '_'), type: f.type || '', data }); transfer.push(data); }
    const r = await act('importFiles', { files: out }, transfer);
    if (r.ok) toast(`已导入 ${files.length} 项素材 · 下次出片由素材剪辑自动编排`, '入');
  } catch (e) { toast('读取文件失败：' + (e && e.message ? e.message : e), '卷', null, null, true); }
  finally { $('#dropZone').classList.remove('busy'); }
}
