function relTime(iso) {
  const t = Date.parse(iso); if (!Number.isFinite(t)) return '';
  const d = (Date.now() - t) / 1000;
  if (d < 90) return '刚刚'; if (d < 3600) return Math.round(d / 60) + ' 分钟前'; if (d < 86400) return Math.round(d / 3600) + ' 小时前'; if (d < 86400 * 30) return Math.round(d / 86400) + ' 天前';
  return new Date(t).toLocaleDateString('zh-CN');
}

/* 顶栏宽度不足时依次收起：状态灯文字 → 品牌字 → 页签英文/计数 → 镜头文字 → 状态灯 → 工程名缩短（永不重叠） */
function fitTopbar() {
  const tb = $('#topbar'); if (!tb) return;
  const steps = ['tb-1', 'tb-2', 'tb-3', 'tb-4', 'tb-5', 'tb-6'];
  steps.forEach((s) => tb.classList.remove(s));
  const over = () => tb.scrollWidth > tb.clientWidth + 1 || $('#tbFill').getBoundingClientRect().width < 4;
  for (const s of steps) { if (!over()) break; tb.classList.add(s); }
}
